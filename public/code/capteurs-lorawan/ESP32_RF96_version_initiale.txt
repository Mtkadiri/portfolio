#include <lmic.h>
#include <hal/hal.h>
#include <SPI.h>
#include "DHT.h"

#define DHTPIN 27
#define DHTTYPE DHT11

DHT dht(DHTPIN, DHTTYPE);

// =====================
// SPI pins
// =====================
#define SCK_PIN   18
#define MISO_PIN  19
#define MOSI_PIN  23
#define NSS_PIN   5
#define RST_PIN   14
#define DIO0_PIN  2
#define DIO1_PIN  4   // new wire added here

// =====================
// LMIC pin mapping
// =====================
const lmic_pinmap lmic_pins = {
    .nss = NSS_PIN,
    .rxtx = LMIC_UNUSED_PIN,
    .rst = RST_PIN,
    .dio = {DIO0_PIN, DIO1_PIN, LMIC_UNUSED_PIN}
};

// =====================
// ABP keys
// =====================
static const u4_t DEVADDR = 0x00000000; // À personnaliser

static const u1_t NWKSKEY[16] = {0}; // Remplacer par votre clé personnelle (16 octets)

static const u1_t APPSKEY[16] = {0}; // Remplacer par votre clé personnelle (16 octets)

// =====================
// Required by this LMIC build
// =====================
void os_getArtEui(u1_t* buf) { }
void os_getDevEui(u1_t* buf) { }
void os_getDevKey(u1_t* buf) { }

// =====================
// Payload / jobs
// =====================
static uint8_t payload[2];
static int counter = 0;
static osjob_t sendjob;

void do_send(osjob_t* j);

// =====================
// Events
// =====================
void onEvent(ev_t ev) {
    switch (ev) {
        case EV_TXCOMPLETE:
            Serial.println("EV_TXCOMPLETE");

            if (LMIC.txrxFlags & TXRX_ACK) {
                Serial.println("  -> ACK received");
            }

            if (LMIC.dataLen) {
                Serial.print("  -> Downlink: ");
                for (int i = 0; i < LMIC.dataLen; i++) {
                    Serial.printf("%02X", LMIC.frame[LMIC.dataBeg + i]);
                }
                Serial.println();
            }

            os_setTimedCallback(&sendjob, os_getTime() + sec2osticks(30), do_send);
            break;


        default:
            Serial.print("Event: ");
            Serial.println((unsigned)ev);
            break;
    }
}

// =====================
// Send function
// =====================
void do_send(osjob_t* j) {
    if (LMIC.opmode & OP_TXRXPEND) {
        Serial.println("OP_TXRXPEND, not sending");
        os_setTimedCallback(&sendjob, os_getTime() + sec2osticks(10), do_send);
        return;
    }

    counter++;

    float t = dht.readTemperature();
    float h = dht.readHumidity();

    Serial.print("t = ");
    Serial.println(t);
    Serial.print("h = ");
    Serial.println(h);

    if (isnan(t) || isnan(h)) {
        Serial.println("Erreur DHT11");
        os_setTimedCallback(&sendjob, os_getTime() + sec2osticks(10), do_send);
        return;
    }

    Serial.print("Temp: ");
    Serial.print(t);
    Serial.print(" | Hum: ");
    Serial.println(h);

    payload[0] = (uint8_t)t;
    payload[1] = (uint8_t)h;

    Serial.print("Sending #");
    Serial.print(counter);
    Serial.print(" : ");
    Serial.printf("%02X %02X\n", payload[0], payload[1]);

    LMIC_setTxData2(4, payload, sizeof(payload), 0);
}

// =====================
// Setup
// =====================
void setup() {
    Serial.begin(115200);
    delay(2000);

    Serial.println();
    Serial.println("=== LMIC ESP32 + RF96 ===");

    dht.begin();
    delay(2000);


    Serial.println("1. SPI begin");
    SPI.begin(SCK_PIN, MISO_PIN, MOSI_PIN, NSS_PIN);

    Serial.println("2. Reset pin");
    pinMode(RST_PIN, OUTPUT);
    digitalWrite(RST_PIN, LOW);
    delay(10);
    digitalWrite(RST_PIN, HIGH);
    delay(10);

    Serial.println("3. Before os_init");
    os_init();
    Serial.println("4. After os_init");

    Serial.println("5. Before LMIC_reset");
    LMIC_reset();
    Serial.println("6. After LMIC_reset");

    Serial.println("7. Before session");
    LMIC_setSession(0x1, DEVADDR, (u1_t*)NWKSKEY, (u1_t*)APPSKEY);
    Serial.println("8. After session");

    LMIC_setLinkCheckMode(0);
    LMIC_setAdrMode(0);

    // EU868 channels
    LMIC_setupChannel(0, 867100000, DR_RANGE_MAP(DR_SF12, DR_SF7), BAND_CENTI);
    LMIC_setupChannel(1, 867300000, DR_RANGE_MAP(DR_SF12, DR_SF7), BAND_CENTI);
    LMIC_setupChannel(2, 867500000, DR_RANGE_MAP(DR_SF12, DR_SF7), BAND_CENTI);

    // optional: keep only channel 0 like your earlier tests
    for (int i = 1; i <= 8; i++) {
        LMIC_disableChannel(i);
    }

    LMIC.dn2Dr = DR_SF9;
    LMIC_setDrTxpow(DR_SF7, 14);

    Serial.print("DevAddr: 0x");
    Serial.println(DEVADDR, HEX);

    Serial.println("9. Before send");
    do_send(&sendjob);
    Serial.println("10. Setup done");
}

// =====================
// Loop
// =====================
void loop() {
    os_runloop_once();
    delay(10);
}
